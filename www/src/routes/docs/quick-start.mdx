# Quick start

## `@solidjs/router`

```tsx
import { createRouter } from '@solidjs/router'
import { Loading } from 'solid-js'
import { RouteProgress } from 'solid-route-progress/router'
import 'solid-route-progress/style.css'

const Router = createRouter({ routes: [{ path: '/', component: Home }] })

<Router>
  {(props) => (
    <>
      <RouteProgress />
      <Loading>{props.children}</Loading>
    </>
  )}
</Router>
```

## Start mode

`@solidjs/vite-plugin` with `start: true`, routes from [`filesystem-routing`](https://www.npmjs.com/package/filesystem-routing):

```tsx
// src/App.tsx
import { createRouter } from '@solidjs/router'
import { fileRoutes } from '@solidjs/router/fs'
import { Loading } from 'solid-js'
import { RouteProgress } from 'solid-route-progress/router'
import { pageRoutes } from 'virtual:file-routes'
import './app.css'

const Router = createRouter({ routes: fileRoutes(pageRoutes) })

export default function App() {
  return (
    <Router>
      {(props) => (
        <>
          <RouteProgress />
          <Loading>{props.children}</Loading>
        </>
      )}
    </Router>
  )
}
```

A complete app, themed with Tailwind: [`examples/start`](https://github.com/kecan0406/solid-route-progress/tree/main/examples/start) ([open in StackBlitz](https://stackblitz.com/github/kecan0406/solid-route-progress/tree/main/examples/start?file=src/App.tsx)).

## Without a router

```tsx
import { NavigationProgress } from 'solid-route-progress/navigation'

const Layout = (props) => (
  <>
    <NavigationProgress />
    {props.children}
  </>
)
```

## Driving it yourself

```tsx
import { ProgressProvider, useProgress } from 'solid-route-progress'

<ProgressProvider>
  <RouteProgress />
  {children}
</ProgressProvider>
```

```ts
const progress = useProgress()

await progress.track(fetch('/api/items'))

const release = progress.start()
progress.set(0.6)
release() // release('error') · release('cancel')
```
